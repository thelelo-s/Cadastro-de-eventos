// --- ARQUIVO: Models/Evento.cs ---
namespace MauiAppEvento.Models
{
    public class Evento
    {
        public string Nome { get; set; }
        public DateTime DataInicio { get; set; }
        public DateTime DataTermino { get; set; }
        public int NumeroParticipantes { get; set; }
        public string Local { get; set; }
        public double CustoPorParticipante { get; set; }

        // Propriedade calculada para Duração (TimeSpan)
        public int DuracaoDias
        {
            get
            {
                TimeSpan diferenca = DataTermino - DataInicio;
                return Math.Max(diferenca.Days, 1); // Garante pelo menos 1 dia
            }
        }

        // Propriedade calculada para Custo Total
        public double CustoTotal
        {
            get
            {
                return NumeroParticipantes * CustoPorParticipante;
            }
        }
    }
}
